The Tool tier plus Bubble Tea v2 (Elm architecture: `Init`/`Update`/`View`), Lip Gloss v2 for styling, and Bubbles v2 for stock components (list, table, viewport, textinput, spinner, help/key).

## Gate: do you need a TUI?

| Need | Use | Not |
|---|---|---|
| Ask a few questions | `huh` form, standalone (below) | Bubble Tea |
| Colored or styled one-off output | `lipgloss.Println(style.Render(...))` | Bubble Tea |
| Spinner, progress bar, status lines, tables, trees, boxes | [ux.md](ux.md): huh spinner, `bubbles/progress.ViewAs`, Lip Gloss | Bubble Tea |
| Live, stateful screens: navigable lists, scrolling viewports, panes, dashboards | **Bubble Tea** | |
| A quick form or dashboard where Bubble Tea's verbosity hurts | tview (widget-based) is a reasonable alternative | |

## Architecture

- **The core returns data; the TUI is one way to get input or show output.** The same result must be reachable with flags and no terminal (`--index`, `--yes`, `--json`).
- **Where to draw depends on the kind of TUI:**
  - **Picker** (produces a result, like `pick` below or fzf): draw on **stderr** (`tea.WithOutput(stderr)`) and print the result to stdout after exit, so `x=$(tool)` works. Gate on stdin and stderr being terminals.
  - **Dashboard or viewer** (no result, like `assets/dashboard`): draw on **stdout** (the default) and gate on stdin **and stdout** being terminals. Then `tool | cat` takes the plain path (TSV/`--json`) instead of drawing a TUI onto a terminal while stdout goes into a pipe.
- **Start the TUI only when interactive:** the gate above passes, `--no-tui` is not set, and `CI` is unset. Otherwise take the flag or plain path, or fail with an error naming the flag.
- **The CLI surface is Tool tier:** Kong (root struct for a single command), exit codes, a dotted module path, and testscript for `--help`, `--no-tui`, `--json` and bad flags all apply here too.
- `tea.WithContext(ctx)` so Ctrl-C (signal) and the program share cancellation.
- `NO_COLOR` and the color profile: Bubble Tea v2 detects the profile (via `colorprofile`) and downsamples Lip Gloss output. You don't need to handle it.

## v1 → v2 gotchas

Most examples online (and in training data) are v1. These lines mark a v1 snippet; translate before using it. Source: the bubbletea and lipgloss `UPGRADE_GUIDE_V2.md`, read 2026-10-09.

| v1 | v2 |
|---|---|
| `github.com/charmbracelet/bubbletea` | `charm.land/bubbletea/v2` (same for lipgloss, bubbles, huh) |
| `View() string` | `View() tea.View`, built with `tea.NewView(s)` |
| `tea.WithAltScreen()`, `tea.EnterAltScreen` | `v.AltScreen = true` on the returned View (also `MouseMode`, `WindowTitle`, `Cursor`) |
| `case tea.KeyMsg:` | `case tea.KeyPressMsg:` (`KeyMsg` is now an interface that also matches releases) |
| `msg.Type == tea.KeyEnter`, `msg.Runes` | `msg.Code == tea.KeyEnter`, `msg.Text` (string), `msg.Mod` |
| `" "` for space | `"space"` |
| `tea.KeyCtrlC` | `msg.String() == "ctrl+c"` |
| `tea.MouseMsg` struct | `MouseClickMsg`, `MouseReleaseMsg`, `MouseWheelMsg`, `MouseMotionMsg` |
| `p.Start()` | `p.Run()` |
| `tea.Sequentially` | `tea.Sequence` |
| `lipgloss.Color` is a type, `AdaptiveColor` | `lipgloss.Color(...)` returns `color.Color`; use `LightDark(isDark)` (`compat.AdaptiveColor` as a stopgap) |
| `lipgloss.NewRenderer` | gone; styles are plain values, and downsampling happens at print time |
| bubbles `m.Width = 40` | `m.SetWidth(40)` / `m.Width()` |

```go
// Command pick lets the user choose one of its arguments and prints the choice to stdout.
// The TUI draws on stderr, so `choice=$(pick a b c)` works like fzf.
package main

import (
	"context"
	"errors"
	"fmt"
	"io"
	"os"
	"os/signal"
	"syscall"

	tea "charm.land/bubbletea/v2"
	"github.com/alecthomas/kong"
	"golang.org/x/term"
)

type CLI struct {
	Items   []string `arg:"" help:"Items to choose from."`
	Index   int      `help:"Pick this index without the TUI (non-interactive)." default:"-1"`
	NoTUI   bool     `name:"no-tui" help:"Never start the TUI."`
	Confirm bool     `help:"Ask for confirmation before printing."`
}

func main() {
	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	code := run(ctx, os.Args[1:], os.Stdin, os.Stdout, os.Stderr)
	stop()
	os.Exit(code)
}

func run(ctx context.Context, args []string, stdin io.Reader, stdout, stderr io.Writer) int {
	var cli CLI
	parser, err := kong.New(&cli, kong.Name("pick"), kong.Writers(stdout, stderr), kong.UsageOnError())
	if err != nil {
		fmt.Fprintf(stderr, "pick: %v\n", err)
		return 1
	}
	if _, err := parser.Parse(args); err != nil {
		fmt.Fprintf(stderr, "pick: %v\n", err)
		return 2
	}

	choice, err := choose(ctx, cli, stdin, stderr)
	if err != nil {
		fmt.Fprintf(stderr, "pick: %v\n", err)
		return 1
	}
	if choice == "" {
		return 1 // user cancelled
	}
	if cli.Confirm && interactive(stdin, stderr) {
		ok, err := confirm(fmt.Sprintf("Use %q?", choice))
		if err != nil || !ok {
			return 1
		}
	}
	fmt.Fprintln(stdout, choice)
	return 0
}

// choose returns the picked item; the TUI is only one way to get it.
func choose(ctx context.Context, cli CLI, stdin io.Reader, tty io.Writer) (string, error) {
	if cli.Index >= 0 {
		if cli.Index >= len(cli.Items) {
			return "", fmt.Errorf("--index %d out of range (have %d items)", cli.Index, len(cli.Items))
		}
		return cli.Items[cli.Index], nil
	}
	if cli.NoTUI || !interactive(stdin, tty) {
		return "", errors.New("not a terminal: pass --index N")
	}
	p := tea.NewProgram(newModel(cli.Items), tea.WithContext(ctx), tea.WithInput(stdin), tea.WithOutput(tty))
	final, err := p.Run()
	if err != nil {
		return "", fmt.Errorf("tui: %w", err)
	}
	return final.(model).chosen, nil
}

// interactive reports whether both ends are terminals and nothing asked for plain mode.
func interactive(in io.Reader, out io.Writer) bool {
	if os.Getenv("CI") != "" {
		return false
	}
	return isTerminal(in) && isTerminal(out)
}

func isTerminal(v any) bool {
	f, ok := v.(*os.File)
	return ok && term.IsTerminal(int(f.Fd()))
}
```

```

> beta
  gamma
↑/↓ move • enter pick • q quit
```

On enter, `View` returns an empty view (the `done` field), so the frame is erased rather than left on screen, and only `beta` is printed to stdout. Verified in a pty: stdout was exactly `beta`, so `x=$(pick alpha beta gamma)` captures it. Piped or in CI, it never draws:

```go
package main

import (
	"strings"

	tea "charm.land/bubbletea/v2"
	"charm.land/lipgloss/v2"
)

var (
	cursorStyle = lipgloss.NewStyle().Foreground(lipgloss.Color("212")).Bold(true)
	helpStyle   = lipgloss.NewStyle().Faint(true)
)

type model struct {
	items  []string
	cursor int
	chosen string
	done   bool // set on quit so the last frame is cleared, not left on screen
}

func newModel(items []string) model { return model{items: items} }

func (m model) Init() tea.Cmd { return nil }

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.KeyPressMsg:
		switch msg.String() {
		case "up", "k":
			if m.cursor > 0 {
				m.cursor--
			}
		case "down", "j":
			if m.cursor < len(m.items)-1 {
				m.cursor++
			}
		case "enter", "space":
			m.chosen = m.items[m.cursor]
			m.done = true
			return m, tea.Quit
		case "q", "esc", "ctrl+c":
			m.done = true
			return m, tea.Quit
		}
	}
	return m, nil
}

func (m model) View() tea.View {
	if m.done {
		return tea.NewView("")
	}
	var b strings.Builder
	for i, item := range m.items {
		if i == m.cursor {
			b.WriteString(cursorStyle.Render("> " + item))
		} else {
			b.WriteString("  " + item)
		}
		b.WriteString("\n")
	}
	b.WriteString(helpStyle.Render("↑/↓ move • enter pick • q quit"))
	return tea.NewView(b.String())
}
```

### Bubbles v2 component APIs (verified with `go doc`, bubbles v2.2.1)

| Component | Construct | Size and content | Notes |
|---|---|---|---|
| `table` | `table.New(table.WithColumns(cols), table.WithFocused(true), table.WithStyles(table.Styles{Header, Cell, Selected}))` | `SetColumns`, `SetRows([]table.Row)`, `SetWidth`, `SetHeight` (**includes the header row**), `Cursor()`/`SetCursor` | `m.table, cmd = m.table.Update(msg)` returns `(table.Model, tea.Cmd)`, not `tea.Model`. It handles keys only while focused (`Focus()`/`Blur()`) |
| `viewport` | `viewport.New()` (no args) | `SetWidth`, `SetHeight`, `SetContent`, `GotoTop()` | `SoftWrap = true` field for wrapping |
| `textinput` | `textinput.New()`; set `Prompt`, `Placeholder` | `Value()`, `Reset()` | `Focus()` **returns a `tea.Cmd`**; return it. `SetVirtualCursor(true)` draws the cursor inside the view |
| `help` + `key` | `help.New()`; `key.NewBinding(key.WithKeys("up","k"), key.WithHelp("↑/k","up"))` | `help.SetWidth`, `help.View(km)` | `km` must implement `ShortHelp() []key.Binding` and `FullHelp() [][]key.Binding`. A `type helpKeys []key.Binding` adapter lets each mode show its own keys. Match with `key.Matches(msg, binding)` |

### Patterns `model.go` gets right

- **Size first:**
  - Store `tea.WindowSizeMsg` and call one `layout()` that sizes every component.
  - `View()` returns `tea.NewView("")` until the first size arrives.
  - Set `v.AltScreen = true` on the returned view for full-screen apps.
- **Budget the height explicitly:** `const chromeLines = title + detail box content + 2 border + status + help`. Then `table.SetHeight(height - chromeLines)`. Add a test that `lipgloss.Height(m.View().Content) <= height`; an off-by-one scrolls the whole frame.
- **Refresh by chaining, not a free-running ticker:** `Init` returns the first scan `Cmd`. Handling its result message returns `tea.Tick(interval, …refreshMsg)`, and `refreshMsg` starts the next scan. Scans never overlap, and a slow scan just stretches the interval.
- **Modes:** in filter mode, route every key to the textinput first (typing `q` must not quit; tested). Only `ctrl+c`, `enter` and `esc` are intercepted. Blur the table while filtering and focus it again after.
- **Keep the selection across refreshes:** remember the selected item's identity before rebuilding rows, then restore the cursor to it.
- **Work goes in `tea.Cmd`s**, never in `Update`: `scanCmd(root)` returns a func that does the I/O and returns a message.

```go
package main

import (
	"fmt"
	"strconv"
	"strings"
	"time"

	"charm.land/bubbles/v2/help"
	"charm.land/bubbles/v2/key"
	"charm.land/bubbles/v2/table"
	"charm.land/bubbles/v2/textinput"
	"charm.land/bubbles/v2/viewport"
	tea "charm.land/bubbletea/v2"
	"charm.land/lipgloss/v2"

	"github.com/jz-wilson/portview/internal/scan"
)

const (
	detailLines = 4 // content lines inside the detail box
	// Rows the table does not own: title, detail box (content + 2 border), status, help.
	chromeLines = 1 + detailLines + 2 + 1 + 1
)

var (
	accent     = lipgloss.Color("212")
	titleStyle = lipgloss.NewStyle().Bold(true).Foreground(accent)
	faintStyle = lipgloss.NewStyle().Faint(true)
	boxStyle   = lipgloss.NewStyle().Border(lipgloss.RoundedBorder()).BorderForeground(lipgloss.Color("240")).Padding(0, 1)
)

// scannedMsg carries one scan result; refreshMsg asks for the next one.
type scannedMsg struct {
	at    time.Time
	socks []scan.Socket
	err   error
}

type refreshMsg struct{}

type keyMap struct {
	Up, Down, Filter, Accept, Clear, Quit key.Binding
}

func newKeyMap() keyMap {
	return keyMap{
		Up:     key.NewBinding(key.WithKeys("up", "k"), key.WithHelp("↑/k", "up")),
		Down:   key.NewBinding(key.WithKeys("down", "j"), key.WithHelp("↓/j", "down")),
		Filter: key.NewBinding(key.WithKeys("/"), key.WithHelp("/", "filter")),
		Accept: key.NewBinding(key.WithKeys("enter"), key.WithHelp("enter", "apply")),
		Clear:  key.NewBinding(key.WithKeys("esc"), key.WithHelp("esc", "clear")),
		Quit:   key.NewBinding(key.WithKeys("q", "ctrl+c"), key.WithHelp("q", "quit")),
	}
}

// helpKeys adapts a binding list to help.KeyMap, so each mode shows its own keys.
type helpKeys []key.Binding

func (h helpKeys) ShortHelp() []key.Binding  { return h }
func (h helpKeys) FullHelp() [][]key.Binding { return [][]key.Binding{h} }

type model struct {
	root     string
	interval time.Duration

	socks   []scan.Socket // latest scan
	visible []scan.Socket // socks after the filter; row i of the table is visible[i]
	err     error
	at      time.Time

	table     table.Model
	detail    viewport.Model
	filter    textinput.Model
	filtering bool
	help      help.Model
	keys      keyMap

	width, height int
}

func newModel(root string, interval time.Duration) model {
	t := table.New(
		table.WithColumns(columns(80)),
		table.WithFocused(true),
		table.WithStyles(table.Styles{
			Header:   titleStyle.Padding(0, 1),
			Cell:     lipgloss.NewStyle().Padding(0, 1),
			Selected: lipgloss.NewStyle().Bold(true).Foreground(accent),
		}),
	)
	f := textinput.New()
	f.Prompt = "/ "
	f.Placeholder = "port, address, pid, process"
	f.SetVirtualCursor(true)
	d := viewport.New()
	d.SoftWrap = true
	return model{root: root, interval: interval, table: t, detail: d, filter: f, help: help.New(), keys: newKeyMap()}
}

func scanCmd(root string) tea.Cmd {
	return func() tea.Msg {
		socks, err := scan.Listening(root)
		return scannedMsg{at: time.Now(), socks: socks, err: err}
	}
}

func (m model) Init() tea.Cmd { return scanCmd(m.root) }

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		m.width, m.height = msg.Width, msg.Height
		m.layout()
		return m, nil

	case scannedMsg:
		m.err, m.at = msg.err, msg.at
		if msg.err == nil {
			m.socks = msg.socks
			m.rebuild()
		}
		// Schedule the next scan only after this one lands, so scans never pile up.
		return m, tea.Tick(m.interval, func(time.Time) tea.Msg { return refreshMsg{} })

	case refreshMsg:
		return m, scanCmd(m.root)

	case tea.KeyPressMsg:
		if m.filtering {
			return m.updateFilter(msg)
		}
		switch {
		case key.Matches(msg, m.keys.Quit):
			return m, tea.Quit
		case key.Matches(msg, m.keys.Filter):
			m.filtering = true
			m.table.Blur()
			return m, m.filter.Focus()
		case key.Matches(msg, m.keys.Clear):
			m.filter.Reset()
			m.rebuild()
			return m, nil
		}
		var cmd tea.Cmd
		m.table, cmd = m.table.Update(msg)
		m.syncDetail()
		return m, cmd
	}
	return m, nil
}

func (m model) updateFilter(msg tea.KeyPressMsg) (tea.Model, tea.Cmd) {
	switch {
	case msg.String() == "ctrl+c":
		return m, tea.Quit
	case key.Matches(msg, m.keys.Accept), key.Matches(msg, m.keys.Clear):
		if key.Matches(msg, m.keys.Clear) {
			m.filter.Reset()
		}
		m.filtering = false
		m.filter.Blur()
		m.table.Focus()
		m.rebuild()
		return m, nil
	}
	var cmd tea.Cmd
	m.filter, cmd = m.filter.Update(msg)
	m.rebuild() // live filtering
	return m, cmd
}

// rebuild reapplies the filter to the latest scan and keeps the cursor on the same socket.
func (m *model) rebuild() {
	var selected *scan.Socket
	if s, ok := m.selected(); ok {
		selected = &s
	}
	m.visible = m.visible[:0:0]
	q := strings.ToLower(strings.TrimSpace(m.filter.Value()))
	rows := make([]table.Row, 0, len(m.socks))
	for _, s := range m.socks {
		row := table.Row{strconv.Itoa(s.Port), s.Addr, pidText(s), processText(s)}
		if q != "" && !strings.Contains(strings.ToLower(strings.Join(row, "\x00")+"\x00"+s.Cmdline), q) {
			continue
		}
		m.visible = append(m.visible, s)
		rows = append(rows, row)
	}
	m.table.SetRows(rows)
	cursor := min(m.table.Cursor(), max(len(rows)-1, 0))
	for i, s := range m.visible {
		if selected != nil && s.Proto == selected.Proto && s.Addr == selected.Addr && s.Port == selected.Port {
			cursor = i
		}
	}
	m.table.SetCursor(cursor)
	m.syncDetail()
}

func (m model) selected() (scan.Socket, bool) {
	i := m.table.Cursor()
	if i < 0 || i >= len(m.visible) {
		return scan.Socket{}, false
	}
	return m.visible[i], true
}

func (m *model) syncDetail() {
	m.detail.SetContent(detailText(m.selected()))
	m.detail.GotoTop()
}

func detailText(s scan.Socket, ok bool) string {
	if !ok {
		return "No listening sockets match."
	}
	cmd := s.Cmdline
	switch {
	case s.PID == 0:
		cmd = "(owner not inspectable: run as the same user or root)"
	case cmd == "":
		cmd = "(empty: kernel thread or hidden cmdline)"
	}
	return fmt.Sprintf("Port     %d/%s\nAddress  %s\nProcess  %s (pid %s)\nCommand  %s",
		s.Port, s.Proto, s.Addr, processText(s), pidText(s), cmd)
}

// columns sizes the table to the terminal; ADDRESS takes whatever the fixed columns leave.
func columns(width int) []table.Column {
	const port, pid, proc, padding = 6, 7, 20, 2 * 4
	addr := max(width-port-pid-proc-padding, 12)
	return []table.Column{
		{Title: "PORT", Width: port},
		{Title: "ADDRESS", Width: addr},
		{Title: "PID", Width: pid},
		{Title: "PROCESS", Width: proc},
	}
}

func (m *model) layout() {
	m.table.SetColumns(columns(m.width))
	m.table.SetWidth(m.width)
	m.table.SetHeight(max(m.height-chromeLines, 3))
	m.detail.SetWidth(max(m.width-4, 10)) // box border (2) + padding (2)
	m.detail.SetHeight(detailLines)
	m.help.SetWidth(m.width)
}

func (m model) View() tea.View {
	if m.width == 0 {
		return tea.NewView("")
	}
	title := titleStyle.Render("portview") + faintStyle.Render(fmt.Sprintf("  listening TCP · refresh %s", m.interval))
	box := boxStyle.Width(m.width).Render(m.detail.View())

	var status string
	switch {
	case m.err != nil:
		status = prefix(LevelError, true) + " " + m.err.Error()
	case m.filtering:
		status = m.filter.View()
	case m.filter.Value() != "":
		status = faintStyle.Render(fmt.Sprintf("filter %q: %d of %d", m.filter.Value(), len(m.visible), len(m.socks)))
	default:
		status = faintStyle.Render(fmt.Sprintf("%d sockets · updated %s", len(m.socks), m.at.Format("15:04:05")))
	}

	keys := helpKeys{m.keys.Up, m.keys.Down, m.keys.Filter, m.keys.Clear, m.keys.Quit}
	if m.filtering {
		keys = helpKeys{m.keys.Accept, m.keys.Clear}
	}
	v := tea.NewView(lipgloss.JoinVertical(lipgloss.Left, title, m.table.View(), box, status, m.help.View(keys)))
	v.AltScreen = true
	return v
}
```

## Prompts without a TUI (huh)

`huh` runs standalone, without a Bubble Tea program of your own. Use it in Tool-tier CLIs for confirmations and small forms. Gate it exactly like the TUI (`interactive(...)`), and keep a flag for every value it asks for.

<!-- asset: tui/confirm.go -->
```go
package main

import (
	"os"

	"charm.land/huh/v2"
)

// confirm asks a yes/no question. ACCESSIBLE=1 swaps the TUI for plain line prompts.
func confirm(title string) (bool, error) {
	var ok bool
	err := huh.NewForm(huh.NewGroup(
		huh.NewConfirm().Title(title).Value(&ok),
	)).WithAccessible(os.Getenv("ACCESSIBLE") != "").Run()
	return ok, err
}
```

`WithAccessible(true)` swaps the TUI for plain line prompts that work with screen readers. Turn it on from an env var or flag, never by default.

## Tests

1. **Pure `Update` tests first.** A model is a value: send `tea.KeyPressMsg{Code: tea.KeyDown}` and assert on the returned model. They're fast, need no terminal, and cover most logic. To **type text** into a textinput, send `tea.KeyPressMsg{Code: r, Text: string(r)}`; `Code` alone moves nothing into the input. In teatest, `tm.Type("8799")` does this for you.
2. **teatest v2 for the real loop:** a handful of tests, not one per key. It lives in `x/exp`, is untagged and experimental, so `go get` pins a pseudo-version: keep it pinned and update it on purpose. `tm.FinalOutput` + `teatest.RequireEqualOutput` gives golden files (`go test -update` rewrites them). Set `teatest.WithInitialTermSize` so output is deterministic. `WaitFor` must match text that is actually on screen, not text in a pane for an unselected row. Its 1 s default is tight under `-race`, so pass `teatest.WithDuration(5*time.Second)` as `WaitFor`'s last argument. Use `FinalModel` to assert on state.
4. **Look at the real thing before calling it done.** Run it in tmux and capture the screen:
   ```bash
   tmux new-session -d -s t -x 100 -y 24 ./tool; sleep 1
   tmux send-keys -t t / ; tmux send-keys -t t -l 'query'; tmux send-keys -t t Enter; sleep 1
   tmux capture-pane -t t -p        # the rendered frame, as text
   tmux send-keys -t t q            # and check it quits
   ```
   tmux renders Bubble Tea v2 correctly, including the alt screen and kitty keyboard sequences. `script` + pyte needs preprocessing (cut at `ESC[?1049l`, strip `ESC[>…u`), so prefer tmux for full-screen apps.
3. **The non-TTY path:** call `run` with `bytes.Buffer`s. A buffer is never a terminal, so this exercises the fallback for free.

```go
package main

import (
	"bytes"
	"context"
	"strings"
	"testing"
	"time"

	tea "charm.land/bubbletea/v2"
	"github.com/charmbracelet/x/exp/teatest/v2"
)

// Update is a pure function: drive it with messages, assert on the model.
func TestUpdateMovesAndPicks(t *testing.T) {
	var m tea.Model = newModel([]string{"a", "b", "c"})
	for _, k := range []tea.KeyPressMsg{{Code: tea.KeyDown}, {Code: tea.KeyDown}, {Code: tea.KeyUp}} {
		m, _ = m.Update(k)
	}
	m, cmd := m.Update(tea.KeyPressMsg{Code: tea.KeyEnter})
	if got := m.(model).chosen; got != "b" {
		t.Fatalf("chosen = %q, want b", got)
	}
	if cmd == nil {
		t.Fatal("enter should return tea.Quit")
	}
}

// teatest runs the real program loop against a fake terminal.
func TestProgramPicks(t *testing.T) {
	tm := teatest.NewTestModel(t, newModel([]string{"a", "b"}), teatest.WithInitialTermSize(40, 10))
	teatest.WaitFor(t, tm.Output(), func(b []byte) bool { return bytes.Contains(b, []byte("> a")) })
	tm.Send(tea.KeyPressMsg{Code: tea.KeyDown})
	tm.Send(tea.KeyPressMsg{Code: tea.KeyEnter})
	final := tm.FinalModel(t, teatest.WithFinalTimeout(2*time.Second))
	if got := final.(model).chosen; got != "b" {
		t.Fatalf("chosen = %q, want b", got)
	}
}

// The non-TTY path must work without a terminal: a bytes.Buffer is never one.
func TestNonInteractive(t *testing.T) {
	var out, errb bytes.Buffer
	if code := run(context.Background(), []string{"--index", "1", "x", "y"}, strings.NewReader(""), &out, &errb); code != 0 {
		t.Fatalf("exit %d: %s", code, errb.String())
	}
	if out.String() != "y\n" {
		t.Fatalf("stdout = %q", out.String())
	}
	out.Reset()
	errb.Reset()
	if code := run(context.Background(), []string{"x", "y"}, strings.NewReader(""), &out, &errb); code != 1 || !strings.Contains(errb.String(), "--index") {
		t.Fatalf("want exit 1 naming --index, got %d: %s", code, errb.String())
	}
}
```
